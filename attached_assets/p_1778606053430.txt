## 📋 PART 5: Multi-RPC Implementation (internal/rpc/multi_rpc.go)

```go
package rpc

import (
    "context"
    "sync"
    "time"
    "math/rand"
    "github.com/ethereum/go-ethereum/ethclient"
    "github.com/ethereum/go-ethereum/rpc"
)

type MultiRPCClient struct {
    primaries   []string
    backups     []string
    clients     map[string]*ethclient.Client
    currentIdx  int
    mu          sync.RWMutex
    healthy     map[string]bool
}

func NewMultiRPCClient(primaries, backups []string) *MultiRPCClient {
    m := &MultiRPCClient{
        primaries:  primaries,
        backups:    backups,
        clients:    make(map[string]*ethclient.Client),
        healthy:    make(map[string]bool),
        currentIdx: 0,
    }
    go m.healthChecker()
    return m
}

func (m *MultiRPCClient) healthChecker() {
    ticker := time.NewTicker(30 * time.Second)
    for range ticker.C {
        allEndpoints := append(m.primaries, m.backups...)
        for _, endpoint := range allEndpoints {
            client, err := rpc.Dial(endpoint)
            if err != nil {
                m.healthy[endpoint] = false
                continue
            }
            var result bool
            err = client.CallContext(context.Background(), &result, "net_listening")
            m.healthy[endpoint] = err == nil
            client.Close()
        }
    }
}

func (m *MultiRPCClient) GetClient() (*ethclient.Client, error) {
    m.mu.Lock()
    defer m.mu.Unlock()
    
    endpoints := append(m.primaries, m.backups...)
    startIdx := m.currentIdx
    
    for i := 0; i < len(endpoints); i++ {
        idx := (startIdx + i) % len(endpoints)
        endpoint := endpoints[idx]
        
        if !m.healthy[endpoint] {
            continue
        }
        
        if client, exists := m.clients[endpoint]; exists {
            m.currentIdx = (idx + 1) % len(endpoints)
            return client, nil
        }
        
        client, err := ethclient.Dial(endpoint)
        if err == nil {
            m.clients[endpoint] = client
            m.currentIdx = (idx + 1) % len(endpoints)
            return client, nil
        }
    }
    return nil, fmt.Errorf("no healthy RPC endpoints available")
}

func (m *MultiRPCClient) GetRandomHealthyEndpoint() string {
    endpoints := append(m.primaries, m.backups...)
    healthy := []string{}
    for _, ep := range endpoints {
        if m.healthy[ep] {
            healthy = append(healthy, ep)
        }
    }
    if len(healthy) == 0 {
        return endpoints[0]
    }
    return healthy[rand.Intn(len(healthy))]
}
```


## 📋 PART 6: AI Orchestrator (internal/ai/orchestrator.go)

```go
package ai

import (
    "context"
    "encoding/json"
    "sync"
    "time"
)

type AIOrchestrator struct {
    gemini     *GeminiClient
    groq       *GroqClient
    huangfing  *HuangfingClient
    weights    map[string]float64
    mu         sync.RWMutex
}

type TradingDecision struct {
    Action        string  `json:"action"` // BUY, SELL, HOLD
    Confidence    float64 `json:"confidence"`
    TokenAddress  string  `json:"token_address"`
    AmountUSD     float64 `json:"amount_usd"`
    Reasoning     string  `json:"reasoning"`
    ProviderVotes map[string]string `json:"provider_votes"`
}

func (ai *AIOrchestrator) GetTradingDecision(tokenAddress string, marketData map[string]interface{}) *TradingDecision {
    ctx, cancel := context.WithTimeout(context.Background(), 500*time.Millisecond)
    defer cancel()
    
    resultChan := make(chan map[string]*TradingDecision, 3)
    
    // Run all 3 AI providers in parallel
    go func() {
        resultChan <- ai.gemini.Analyze(ctx, tokenAddress, marketData)
    }()
    go func() {
        resultChan <- ai.groq.Analyze(ctx, tokenAddress, marketData)
    }()
    go func() {
        resultChan <- ai.huangfing.Analyze(ctx, tokenAddress, marketData)
    }()
    
    results := make(map[string]*TradingDecision)
    for i := 0; i < 3; i++ {
        select {
        case res := <-resultChan:
            for provider, decision := range res {
                results[provider] = decision
            }
        case <-ctx.Done():
            // Timeout, continue with available results
        }
    }
    
    return ai.consolidateResults(results)
}

func (ai *AIOrchestrator) consolidateResults(results map[string]*TradingDecision) *TradingDecision {
    actionVotes := make(map[string]float64)
    totalConfidence := 0.0
    
    for provider, decision := range results {
        weight := ai.weights[provider]
        actionVotes[decision.Action] += weight * (decision.Confidence / 100.0)
        totalConfidence += decision.Confidence * weight
    }
    
    finalAction := "HOLD"
    maxVotes := 0.0
    for action, votes := range actionVotes {
        if votes > maxVotes {
            maxVotes = votes
            finalAction = action
        }
    }
    
    return &TradingDecision{
        Action:        finalAction,
        Confidence:    totalConfidence,
        ProviderVotes: make(map[string]string),
    }
}
```
